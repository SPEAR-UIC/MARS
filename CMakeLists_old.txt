cmake_minimum_required(VERSION 3.20)
project(cqsimcpp LANGUAGES CXX C)

# Tells CMake to do a debug build (adds -g automatically)
set(CMAKE_BUILD_TYPE Debug)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# ==========================================
# SANITIZER OPTIONS (OFF by default)
# ==========================================
option(USE_ASAN "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(USE_TSAN "Enable ThreadSanitizer" OFF)

# Enforce the rule: Cannot use both at once
if(USE_ASAN AND USE_TSAN)
    message(FATAL_ERROR "AddressSanitizer and ThreadSanitizer cannot be used together! Please enable only one.")
endif()

set(SANITIZER_FLAGS "")
if(USE_ASAN)
    message(STATUS "Building with AddressSanitizer (ASan) & UBSan")
    # -fno-omit-frame-pointer makes the crash backtraces much easier to read
    set(SANITIZER_FLAGS "-fsanitize=address" "-fsanitize=undefined" "-fno-omit-frame-pointer")
elseif(USE_TSAN)
    message(STATUS "Building with ThreadSanitizer (TSan)")
    set(SANITIZER_FLAGS "-fsanitize=thread" "-fno-omit-frame-pointer")
endif()
# ==========================================

find_package(OpenMP REQUIRED)

add_executable(cqsimcpp
    src/main.cpp
    src/io/swf_reader.cpp
    src/simulator/logger.cpp
    src/simulator/simulator.cpp
    src/experiment/driver.cpp
    src/experiment/experiment.cpp
    src/experiment/registry.cpp
    src/experiment/experiments/exp1.cpp
    src/experiment/experiments/exp2.cpp
    src/experiment/experiments/exp3.cpp
    src/experiment/experiments/exp4.cpp
    src/experiment/json_experiment.cpp
    src/mcts/mcts.cpp
    src/evaluation/instance_eval.cpp
)

# Apply the chosen sanitizer flags to the compiler and the linker
if(SANITIZER_FLAGS)
    target_compile_options(cqsimcpp PRIVATE ${SANITIZER_FLAGS})
    target_link_options(cqsimcpp PRIVATE ${SANITIZER_FLAGS})
endif()

target_include_directories(cqsimcpp PRIVATE third_party)
target_link_libraries(cqsimcpp PRIVATE OpenMP::OpenMP_CXX)

# ==========================================
# GTEST
# ==========================================
include(FetchContent)
FetchContent_Declare(
    googletest
    GIT_REPOSITORY https://github.com/google/googletest.git
    GIT_TAG        v1.15.2
)
set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(googletest)
enable_testing()

# Helper to register a test executable
# Usage: add_scheduler_test(test_name src1.cpp src2.cpp ...)
function(add_scheduler_test test_name)
    add_executable(${test_name} ${ARGN})
    target_include_directories(${test_name} PRIVATE src2 third_party)
    target_link_libraries(${test_name} PRIVATE GTest::gtest_main)
    include(GoogleTest)
    gtest_discover_tests(${test_name})
endfunction()

# ── Test targets (add one line per test file as you create them) ──────────────
add_scheduler_test(test_resources tests/test_resources.cpp)
